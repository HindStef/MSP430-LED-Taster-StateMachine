158void init_clock(void) {
159 // Config clocks. MCLK=SMCLK=FLL=8MHz; ACLK=REFO=32kHz
160 uint32_t mclkFreq = 8000000 ;
161
162 // Assign the REFO as the FLL reference clock
163 UCS_initClockSignal(
164 UCS_FLLREF,
165 UCS_REFOCLK_SELECT,
166 UCS_CLOCK_DIVIDER_1);
167
168 // Assign the REFO as the source for ACLK
169 UCS_initClockSignal(
170 UCS_ACLK,
171 UCS_REFOCLK_SELECT,
172 UCS_CLOCK_DIVIDER_1);
173
174 UCS_initFLLSettle(
175 mclkFreq/1000,
176 mclkFreq/32768);
177 //use REFO for FLL and ACLK
178 UCSCTL3 = (UCSCTL3 & ~(SELREF_7)) | (SELREF__REFOCLK);
179 UCSCTL4 = (UCSCTL4 & ~(SELA_7)) | (SELA__REFOCLK);
180}

185void setup (void) {
186 StopWatchdog () ;
187 // MSP430 USB requires a Vcore setting of at least 2. 2 is high enough
188 // for 8MHz MCLK, below.
189 PMM_setVCore(PMM_CORE_LEVEL_2);
190
191 init_clock();
192 bcUartInit(); // Init the back-channel UART: 115200 bps
193
194 // initPorts(); // Config all the GPIOS for low-power (output low)
195 // an Port 1 bit 0 ist die rote LED angeschlossen, LED über Vorwiderstand an VCC, 0 bedeutet LED ist an und leuchtet.
196 // an Port 4 bit 7 ist die grüne LED angeschlossen,
197 // an Port 2 bit 1 ist der Taster S1 angeschlossen
198 P1DIR |= 0x01; // P1.0 output, P1.1 input
199 P1OUT = 0x03; // rote LED aus, LED
200 P4DIR = 0x80; // P4.7 output grüne LED / LED_GREEN_P4
201 P4OUT = 0x00;
202 //Set P2.1 to input direction
203 GPIO_setAsInputPin ( GPIO_PORT_P2, GPIO_PIN1 ) ;
204 GPIO_setAsInputPinWithPullUpResistor ( GPIO_PORT_P2, GPIO_PIN1 ) ;
205 GPIO_setAsOutputPin ( GPIO_PORT_P1, GPIO_PIN1 ) ;
206 GPIO_setAsOutputPin ( GPIO_PORT_P4, GPIO_PIN7 ) ;
207
208 GPIO_setOutputLowOnPin(GPIO_PORT_P1, GPIO_PIN1);
209 GPIO_setOutputLowOnPin(GPIO_PORT_P4, GPIO_PIN7);
210
211 // init Variable of FSM
212 Button_Init(&g_button);
213 LED_Init(&g_led);
214}

307static ButtonEvent_t Button_Process_1ms(Button_t *btn) {
308 ButtonEvent_t event = BTN_EVT_NONE;
309 bool raw_level = Button_ReadRaw();
310
311 if (raw_level == true) {
312 P4OUT |= LED_GREEN_P4; // Grüne LED an
313 } else {
314 P4OUT &= ~LED_GREEN_P4; // Grüne LED aus
315 }
316
317 switch (btn->state)
318 {
319 case BTN_STATE_IDLE:
320 if (raw_level == true) {
321 /* Moeglicher Tastendruck erkannt -> entprellen */
322 btn->state = BTN_STATE_DEBOUNCE;
323 btn->debounce_time_ms = 0U;
324 }
325 break;
326
327 case BTN_STATE_DEBOUNCE:
328 btn->debounce_time_ms++;
329 if (raw_level == false) {
330 /* War nur ein Stoerimpuls, zurueck zu IDLE */
331 btn->state = BTN_STATE_IDLE;
332 }
333 else if (btn->debounce_time_ms >= DEBOUNCE_TIME_MS) {
334 /* Tastendruck bestaetigt */
335 btn->state = BTN_STATE_PRESSED;
336 btn->press_time_ms = 0U;
337 str_to_bcUART ("BTN_STATE_PRESSED.\r\n") ;
338 }
339 break;
340
341 case BTN_STATE_PRESSED:
342 if (raw_level == false) {
343 /* Taste innerhalb der kurzen Zeit losgelassen */
344 event = BTN_EVT_SHORT_PRESS;
345 btn->state = BTN_STATE_IDLE;
346 str_to_bcUART ("BTN_EVT_SHORT_PRESS.\r\n") ;
347 } else {
348 btn->press_time_ms++;
349 if (btn->press_time_ms >= LONG_PRESS_THRESHOLD_MS) {
350 /* Schwelle erreicht -> langer Tastendruck melden */
351 event = BTN_EVT_LONG_PRESS;
352 btn->state = BTN_STATE_LONG_DETECTED;
353 str_to_bcUART ("BTN_EVT_LONG_PRESS.\r\n") ;
354 }
355 }
356 break;
357
358 case BTN_STATE_LONG_DETECTED:
359 /* LONG_PRESS wurde bereits gemeldet, warten auf Loslassen */
360 if (raw_level == false) {
361 event = BTN_EVT_RELEASE;
362 btn->state = BTN_STATE_IDLE;
363 str_to_bcUART ("BTN_EVT_RELEASE.\r\n") ;
364 }
365 break;
366
367 default:
368 btn->state = BTN_STATE_IDLE;
369 break;
370 }
371
372 btn->raw_level_last = raw_level;
373
374 return event;
375}

394static void LED_HandleEvent(Led_t *led, ButtonEvent_t event) {
395 switch (event)
396 {
397 case BTN_EVT_SHORT_PRESS:
398 if (led->state == LED_STATE_ON) {
399 led->state = LED_STATE_OFF;
400 LED_SetHardware(false);
401 str_to_bcUART ("rote LED aus.\r\n") ;
402 } else if (led->state == LED_STATE_BLINK) {
403 /* aus BLINK heraus: LED ausschalten */
404 led->state = LED_STATE_OFF;
405 LED_SetHardware(false);
406 str_to_bcUART ("stop blinken und rote LED aus.\r\n") ;
407 } else {
408 /* aus OFF heraus: LED einschalten */
409 led->state = LED_STATE_ON;
410 LED_SetHardware(true);
411 str_to_bcUART ("rote LED an.\r\n") ;
412 }
413 break;
414
415 case BTN_EVT_LONG_PRESS:
416 led->state = LED_STATE_BLINK;
417 led->blink_timer_ms = 0U;
418 led->blink_phase_on = true;
419 LED_SetHardware(true);
420 str_to_bcUART ("start blinken mit der roten LED.\r\n") ;
421 break;
422
423 case BTN_EVT_RELEASE:
424 case BTN_EVT_NONE:
425 default:
426 /* Kein Zustandswechsel bei diesen Ereignissen */
427 break;
428 }
429}

435static void LED_Process_1ms(Led_t *led) {
436 if (led->state == LED_STATE_BLINK) {
437 led->blink_timer_ms++;
438 if (led->blink_timer_ms >= BLINK_PERIOD_MS) {
439 led->blink_timer_ms = 0U;
440 led->blink_phase_on = !led->blink_phase_on;
441 LED_SetHardware(led->blink_phase_on);
442 }
443 }
444 /* OFF und ON benoetigen keine periodische Aktion */
445}
