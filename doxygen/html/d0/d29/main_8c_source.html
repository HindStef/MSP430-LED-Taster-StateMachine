131void init_clock(void) {
132 // Config clocks. MCLK=SMCLK=FLL=8MHz; ACLK=REFO=32kHz
133 uint32_t mclkFreq = 8000000 ;
134
135 // Assign the REFO as the FLL reference clock
136 UCS_initClockSignal(
137 UCS_FLLREF,
138 UCS_REFOCLK_SELECT,
139 UCS_CLOCK_DIVIDER_1);
140
141 // Assign the REFO as the source for ACLK
142 UCS_initClockSignal(
143 UCS_ACLK,
144 UCS_REFOCLK_SELECT,
145 UCS_CLOCK_DIVIDER_1);
146
147 UCS_initFLLSettle(
148 mclkFreq/1000,
149 mclkFreq/32768);
150 //use REFO for FLL and ACLK
151 UCSCTL3 = (UCSCTL3 & ~(SELREF_7)) | (SELREF__REFOCLK);
152 UCSCTL4 = (UCSCTL4 & ~(SELA_7)) | (SELA__REFOCLK);
153}

158void setup (void) {
159 StopWatchdog () ;
160 // MSP430 USB requires a Vcore setting of at least 2. 2 is high enough
161 // for 8MHz MCLK, below.
162 PMM_setVCore(PMM_CORE_LEVEL_2);
163
164 init_clock();
165 bcUartInit(); // Init the back-channel UART: 115200 bps
166
167 // initPorts(); // Config all the GPIOS for low-power (output low)
168 // an Port 1 bit 0 ist die rote LED angeschlossen, LED über Vorwiderstand an VCC, 0 bedeutet LED ist an und leuchtet.
169 // an Port 4 bit 7 ist die grüne LED angeschlossen,
170 // an Port 2 bit 1 ist der Taster S1 angeschlossen
171 P1DIR |= 0x01; // P1.0 output, P1.1 input
172 P1OUT = 0x03; // rote LED aus, LED
173 P4DIR = 0x80; // P4.7 output grüne LED / LED_GREEN_P4
174 P4OUT = 0x00;
175 //Set P2.1 to input direction
176 GPIO_setAsInputPin ( GPIO_PORT_P2, GPIO_PIN1 ) ;
177 GPIO_setAsInputPinWithPullUpResistor ( GPIO_PORT_P2, GPIO_PIN1 ) ;
178 GPIO_setAsOutputPin ( GPIO_PORT_P1, GPIO_PIN1 ) ;
179 GPIO_setAsOutputPin ( GPIO_PORT_P4, GPIO_PIN7 ) ;
180
181 GPIO_setOutputLowOnPin(GPIO_PORT_P1, GPIO_PIN1);
182 GPIO_setOutputLowOnPin(GPIO_PORT_P4, GPIO_PIN7);
183
184 // init Variable of FSM
185 Button_Init(&g_button);
186 LED_Init(&g_led);
187}

280static ButtonEvent_t Button_Process_1ms(Button_t *btn) {
281 ButtonEvent_t event = BTN_EVT_NONE;
282 bool raw_level = Button_ReadRaw();
283
284 if (raw_level == true) {
285 P4OUT |= LED_GREEN_P4; // Grüne LED an
286 } else {
287 P4OUT &= ~LED_GREEN_P4; // Grüne LED aus
288 }
289
290 switch (btn->state)
291 {
292 case BTN_STATE_IDLE:
293 if (raw_level == true) {
294 /* Moeglicher Tastendruck erkannt -> entprellen */
295 btn->state = BTN_STATE_DEBOUNCE;
296 btn->debounce_time_ms = 0U;
297 }
298 break;
299
300 case BTN_STATE_DEBOUNCE:
301 btn->debounce_time_ms++;
302 if (raw_level == false) {
303 /* War nur ein Stoerimpuls, zurueck zu IDLE */
304 btn->state = BTN_STATE_IDLE;
305 }
306 else if (btn->debounce_time_ms >= DEBOUNCE_TIME_MS) {
307 /* Tastendruck bestaetigt */
308 btn->state = BTN_STATE_PRESSED;
309 btn->press_time_ms = 0U;
310 str_to_bcUART ("BTN_STATE_PRESSED.\r\n") ;
311 }
312 break;
313
314 case BTN_STATE_PRESSED:
315 if (raw_level == false) {
316 /* Taste innerhalb der kurzen Zeit losgelassen */
317 event = BTN_EVT_SHORT_PRESS;
318 btn->state = BTN_STATE_IDLE;
319 str_to_bcUART ("BTN_EVT_SHORT_PRESS.\r\n") ;
320 } else {
321 btn->press_time_ms++;
322 if (btn->press_time_ms >= LONG_PRESS_THRESHOLD_MS) {
323 /* Schwelle erreicht -> langer Tastendruck melden */
324 event = BTN_EVT_LONG_PRESS;
325 btn->state = BTN_STATE_LONG_DETECTED;
326 str_to_bcUART ("BTN_EVT_LONG_PRESS.\r\n") ;
327 }
328 }
329 break;
330
331 case BTN_STATE_LONG_DETECTED:
332 /* LONG_PRESS wurde bereits gemeldet, warten auf Loslassen */
333 if (raw_level == false) {
334 event = BTN_EVT_RELEASE;
335 btn->state = BTN_STATE_IDLE;
336 str_to_bcUART ("BTN_EVT_RELEASE.\r\n") ;
337 }
338 break;
339
340 default:
341 btn->state = BTN_STATE_IDLE;
342 break;
343 }
344
345 btn->raw_level_last = raw_level;
346
347 return event;
348}

367static void LED_HandleEvent(Led_t *led, ButtonEvent_t event) {
368 switch (event)
369 {
370 case BTN_EVT_SHORT_PRESS:
371 if (led->state == LED_STATE_ON) {
372 led->state = LED_STATE_OFF;
373 LED_SetHardware(false);
374 str_to_bcUART ("rote LED aus.\r\n") ;
375 } else if (led->state == LED_STATE_BLINK) {
376 /* aus BLINK heraus: LED ausschalten */
377 led->state = LED_STATE_OFF;
378 LED_SetHardware(false);
379 str_to_bcUART ("stop blinken und rote LED aus.\r\n") ;
380 } else {
381 /* aus OFF heraus: LED einschalten */
382 led->state = LED_STATE_ON;
383 LED_SetHardware(true);
384 str_to_bcUART ("rote LED an.\r\n") ;
385 }
386 break;
387
388 case BTN_EVT_LONG_PRESS:
389 led->state = LED_STATE_BLINK;
390 led->blink_timer_ms = 0U;
391 led->blink_phase_on = true;
392 LED_SetHardware(true);
393 str_to_bcUART ("start blinken mit der roten LED.\r\n") ;
394 break;
395
396 case BTN_EVT_RELEASE:
397 case BTN_EVT_NONE:
398 default:
399 /* Kein Zustandswechsel bei diesen Ereignissen */
400 break;
401 }
402}

408static void LED_Process_1ms(Led_t *led) {
409 if (led->state == LED_STATE_BLINK) {
410 led->blink_timer_ms++;
411 if (led->blink_timer_ms >= BLINK_PERIOD_MS) {
412 led->blink_timer_ms = 0U;
413 led->blink_phase_on = !led->blink_phase_on;
414 LED_SetHardware(led->blink_phase_on);
415 }
416 }
417 /* OFF und ON benoetigen keine periodische Aktion */
418}

434void App_1ms_Tick(void) {
435 ButtonEvent_t event = Button_Process_1ms(&g_button);
436
437 if (event != BTN_EVT_NONE) {
438 LED_HandleEvent(&g_led, event);
439 }
440
441 LED_Process_1ms(&g_led);
